cmake_minimum_required(VERSION 3.16)
project(dk64_ra_probe LANGUAGES C CXX)

# Pin the official client runtime, and permit an ignored local checkout for
# offline development. The mod never fetches this at run time.
set(DK64_RA_RCHEEVOS_SOURCE_DIR "" CACHE PATH "Local rcheevos checkout")
if(DK64_RA_RCHEEVOS_SOURCE_DIR)
    set(rcheevos_SOURCE_DIR "${DK64_RA_RCHEEVOS_SOURCE_DIR}")
else()
    include(FetchContent)
    FetchContent_Declare(rcheevos
        GIT_REPOSITORY https://github.com/RetroAchievements/rcheevos.git
        GIT_TAG 1433173220a7eaede6a9ed7a18e94117be1821e0)
    FetchContent_MakeAvailable(rcheevos)
endif()
if(NOT EXISTS "${rcheevos_SOURCE_DIR}/include/rc_client.h")
    message(FATAL_ERROR "No rcheevos headers at ${rcheevos_SOURCE_DIR}")
endif()
file(GLOB_RECURSE RCHEEVOS_SOURCES CONFIGURE_DEPENDS "${rcheevos_SOURCE_DIR}/src/*.c")
# Only the RetroArch/libretro adapter needs libretro.h; this standalone mod
# uses rc_client directly and must not pull in emulator frontend headers.
list(FILTER RCHEEVOS_SOURCES EXCLUDE REGEX "/rc_libretro\\.c$")
add_library(rcheevos STATIC ${RCHEEVOS_SOURCES})
target_include_directories(rcheevos PUBLIC "${rcheevos_SOURCE_DIR}/include")
set_target_properties(rcheevos PROPERTIES POSITION_INDEPENDENT_CODE ON)

add_library(dk64_ra_probe SHARED probe.cpp memory.cpp client_bridge.cpp guest_catalog.cpp guest_text.cpp badge_cache.cpp http_transport.cpp windows_credentials.cpp diagnostic_log.cpp local_progress.cpp online_awards.cpp toast_sound.cpp)
target_compile_features(dk64_ra_probe PRIVATE cxx_std_17)
target_include_directories(dk64_ra_probe PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(dk64_ra_probe PRIVATE rcheevos)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    find_package(CURL 7.84 REQUIRED)
    find_package(SDL2 REQUIRED)
    target_link_libraries(dk64_ra_probe PRIVATE CURL::libcurl SDL2::SDL2 ${CMAKE_DL_LIBS})
endif()
option(DK64_RA_RAP_DIAGNOSTICS "Inspect DK Rap condition flags in local tracking builds" OFF)
if(DK64_RA_RAP_DIAGNOSTICS)
    target_sources(dk64_ra_probe PRIVATE rap_diagnostics.cpp)
    target_include_directories(dk64_ra_probe PRIVATE "${rcheevos_SOURCE_DIR}/src")
    target_compile_definitions(dk64_ra_probe PRIVATE DK64_RA_RAP_DIAGNOSTICS=1)
endif()
option(DK64_RA_MEMORY_DIAGNOSTICS "Offline-only game-state/address parity checks" OFF)
if(DK64_RA_MEMORY_DIAGNOSTICS)
    target_sources(dk64_ra_probe PRIVATE memory_diagnostics.cpp)
    target_compile_definitions(dk64_ra_probe PRIVATE DK64_RA_MEMORY_DIAGNOSTICS=1)
endif()
if(WIN32)
    target_link_libraries(dk64_ra_probe PRIVATE winhttp credui advapi32 user32 shell32 ole32 winmm)
endif()
set_target_properties(dk64_ra_probe PROPERTIES PREFIX "")
